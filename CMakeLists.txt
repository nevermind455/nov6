cmake_minimum_required(VERSION 3.24)
project(sniper_cpp VERSION 1.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU" AND CMAKE_CXX_COMPILER_VERSION VERSION_LESS 12)
  message(FATAL_ERROR "GCC 12 or later is required")
endif()
option(SNIPER_TESTS "Build native offline tests and benchmarks" ON)
option(SNIPER_SANITIZE "Build with AddressSanitizer and UBSan" OFF)
find_package(Threads REQUIRED)
find_package(OpenSSL 3.0 REQUIRED)
if(POLICY CMP0167)
  # CMake 3.30+ removed FindBoost; use the BoostConfig.cmake from libboost-dev.
  cmake_policy(SET CMP0167 NEW)
endif()
find_package(Boost 1.83.0 EXACT REQUIRED)
find_package(nlohmann_json 3.11.3 EXACT REQUIRED)
find_package(fmt 9.1.0 EXACT REQUIRED)
find_package(simdjson 3.6.4 EXACT REQUIRED)
find_path(NGHTTP2_INCLUDE_DIR nghttp2/nghttp2.h REQUIRED)
find_library(NGHTTP2_LIBRARY NAMES libnghttp2.a nghttp2 REQUIRED)
find_path(SECP256K1_INCLUDE_DIR secp256k1_recovery.h REQUIRED)
find_library(SECP256K1_LIBRARY NAMES libsecp256k1.a secp256k1 REQUIRED)
add_library(sniper_core
  src/config.cpp src/crypto.cpp src/storage.cpp src/book.cpp src/strategy.cpp
  src/pyrt.cpp src/sniper.cpp src/paper.cpp src/settings.cpp
  src/network.cpp src/h2.cpp src/venue.cpp src/runtime.cpp
  src/dashboard.cpp src/dashboard_terminal.cpp third_party/sha3.c)
target_include_directories(sniper_core PUBLIC include ${Boost_INCLUDE_DIRS} ${NGHTTP2_INCLUDE_DIR} ${SECP256K1_INCLUDE_DIR} PRIVATE third_party)
target_compile_definitions(sniper_core PUBLIC BOOST_ERROR_CODE_HEADER_ONLY BOOST_SYSTEM_NO_DEPRECATED)
target_compile_options(sniper_core PUBLIC -O2 -Wall -Wextra -ffp-contract=off)
target_link_libraries(sniper_core PUBLIC OpenSSL::SSL OpenSSL::Crypto Threads::Threads nlohmann_json::nlohmann_json fmt::fmt-header-only simdjson::simdjson ${NGHTTP2_LIBRARY} ${SECP256K1_LIBRARY})
if(SNIPER_SANITIZE)
  target_compile_options(sniper_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(sniper_core PUBLIC -fsanitize=address,undefined)
endif()
add_executable(sniper-cpp src/main.cpp)
add_executable(sniper-replay tools/replay/sniper_replay.cpp)
add_executable(sniper-diff tools/replay/diff_tool.cpp)
target_link_libraries(sniper-diff PRIVATE sniper_core)
set_target_properties(sniper-diff PROPERTIES INSTALL_RPATH "$ORIGIN/lib;$ORIGIN/../lib")
install(TARGETS sniper-diff RUNTIME DESTINATION .)
target_link_libraries(sniper-replay PRIVATE sniper_core)
set_target_properties(sniper-replay PROPERTIES INSTALL_RPATH "$ORIGIN/lib;$ORIGIN/../lib")
install(TARGETS sniper-replay RUNTIME DESTINATION .)
target_link_libraries(sniper-cpp PRIVATE sniper_core)
set_target_properties(sniper-cpp PROPERTIES INSTALL_RPATH "$ORIGIN/lib;$ORIGIN/../lib")
install(TARGETS sniper-cpp RUNTIME DESTINATION .)
if(SNIPER_TESTS)
  find_package(GTest 1.14.0 EXACT REQUIRED)
  find_package(benchmark 1.8.3 EXACT REQUIRED)
  add_executable(sniper-tests tests/native_tests.cpp tests/transport_tests.cpp
    tests/dashboard_tests.cpp tests/discovery_tests.cpp)
  target_link_libraries(sniper-tests PRIVATE sniper_core GTest::gtest_main)
  set_target_properties(sniper-tests PROPERTIES INSTALL_RPATH "$ORIGIN/lib;$ORIGIN/../lib")
  install(TARGETS sniper-tests RUNTIME DESTINATION .)
  target_compile_definitions(sniper-tests PRIVATE FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
  enable_testing()
  add_test(NAME native_suite COMMAND sniper-tests)
  add_executable(sniper-bench tests/benchmark.cpp)
  target_link_libraries(sniper-bench PRIVATE sniper_core benchmark::benchmark_main)
endif()
